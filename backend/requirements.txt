from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from app.api.routes.agents import router as agents_router
from app.api.routes.auth import router as auth_router
from app.api.routes.corpus import router as corpus_router
from app.api.routes.grounding import router as grounding_router
from app.api.routes.llm import router as llm_router
from app.api.routes.production import router as production_router
from app.api.routes.projects import router as projects_router
from app.api.routes.references import router as references_router
from app.api.routes.workflows import router as workflows_router
from app.database import Base, engine

Base.metadata.create_all(bind=engine)

app = FastAPI(
    title="Bedrock AI Addon Orchestrator",
    version="1.1.0",
    description="Plataforma de produção com grounding documental, persistência real, múltiplos agentes e workflow Bedrock.",
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(auth_router, prefix="/api")
app.include_router(projects_router, prefix="/api")
app.include_router(references_router, prefix="/api")
app.include_router(agents_router, prefix="/api")
app.include_router(llm_router, prefix="/api")
app.include_router(corpus_router, prefix="/api")
app.include_router(grounding_router, prefix="/api")
app.include_router(workflows_router, prefix="/api")
app.include_router(production_router, prefix="/api")


@app.get("/api/health")
def health_check():
    return {
        "status": "ok",
        "service": "bedrock-ai-addon-orchestrator",
        "version": "1.1.0",
        "database": "sqlite",
    }
